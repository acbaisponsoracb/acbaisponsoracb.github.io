from fastapi import FastAPI, Request
import httpx

app = FastAPI(title="ACB AI Webhook Service")
ACB_API_URL = "https://sponsoracb.ir/api/v1/chat"

@app.post("/webhook")
async def telegram_webhook(req: Request):
    data = await req.json()
    user_msg = data.get("message", {}).get("text", "")
    
    async with httpx.AsyncClient() as client:
        res = await client.post(ACB_API_URL, json={
            "model": "acb-ultra-v4",
            "prompt": user_msg
        })
    return {"status": "delivered", "reply": res.json()}
